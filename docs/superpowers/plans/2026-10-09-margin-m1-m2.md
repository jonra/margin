# Margin M1 + M2 (Browser + Agent Review) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the MarginCore engine (workspace, syntax, git, search, review) and a native macOS app that can browse Java/Python repos and review agent changes live, including git worktrees.

**Architecture:** One SwiftPM package. Engine modules (`MarginSupport`, `MarginSyntax`, `MarginGit`, `MarginWorkspace`, `MarginSearch`, `MarginReview`) have no UI dependency and are re-exported by `MarginCore`. The `Margin` executable target is a thin SwiftUI/AppKit client. The disk and git are the single source of truth: an FSEvents watcher triggers re-reads, and nothing is updated optimistically.

**Tech Stack:** Swift 6.1 toolchain (Swift 5 language mode), SwiftPM only (no Xcode project), SwiftUI + AppKit (NSTextView), SwiftTreeSitter 0.25.0 with vendored tree-sitter-java v0.23.5 / tree-sitter-python v0.25.0, the git CLI, ripgrep, FSEvents, Swift Testing (`import Testing`).

## Global Constraints

- macOS 14+ (`platforms: [.macOS(.v14)]`). Build with Command Line Tools only: **no Xcode, no `xcodebuild`, no XCTest**. Tests use Swift Testing.
- `swift-tools-version:6.0` with `swiftLanguageModes: [.v5]` (strict-concurrency errors are out of scope for v1).
- Engine targets must not import SwiftUI or AppKit. Only the `Margin` executable target may.
- All git operations go through the git CLI (`Git` struct) with `-C <root>`; parse only machine-readable output.
- External tools are found with `ToolLocator` (PATH plus `/opt/homebrew/bin`, `/usr/local/bin`). A missing tool must surface as `MissingToolError` with an install hint, never as a crash.
- Paths are repo-relative strings inside the engine. Absolute paths are resolved with `realpath` (`String.resolvingRealPath()`) so they match FSEvents and git (`/var` → `/private/var`).
- Do **not** add CodeEditSourceEditor. It fails under SwiftPM without Xcode (a SwiftLint build plugin in its dependency tree). Grammars are vendored because the upstream tree-sitter-python `Package.swift` drops `scanner.c` (link error).
- Bundle id `nl.productlabs.margin`. App name "Margin".
- Every task ends with `swift build` and `swift test` green and a commit. Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

`Sources/MarginSupport/ProcessRunner.swift`:

```swift
import Foundation

public struct ProcessResult: Sendable {
    public let status: Int32
    public let stdout: Data
    public let stderr: Data

    public var stdoutString: String { String(decoding: stdout, as: UTF8.self) }
    public var stderrString: String { String(decoding: stderr, as: UTF8.self) }
}

public struct ProcessError: Error, CustomStringConvertible, Sendable {
    public let command: String
    public let status: Int32
    public let stderr: String

    public init(command: String, status: Int32, stderr: String) {
        self.command = command
        self.status = status
        self.stderr = stderr
    }

    public var description: String { "\(command) failed (\(status)): \(stderr)" }
}

/// Runs external tools (git, rg, …) off the main thread and collects their output.
public enum ProcessRunner {
    public static func run(
        _ executable: URL, _ arguments: [String], cwd: URL? = nil, stdin: Data? = nil
    ) async throws -> ProcessResult {
        try await withCheckedThrowingContinuation { continuation in
            DispatchQueue.global(qos: .userInitiated).async {
                let process = Process()
                process.executableURL = executable
                process.arguments = arguments
                if let cwd { process.currentDirectoryURL = cwd }
                let out = Pipe(), err = Pipe()
                process.standardOutput = out
                process.standardError = err
                let inPipe = stdin == nil ? nil : Pipe()
                process.standardInput = inPipe ?? FileHandle.nullDevice
                do {
                    try process.run()
                } catch {
                    continuation.resume(throwing: error)
                    return
                }
                if let inPipe, let stdin {
                    DispatchQueue.global().async {
                        inPipe.fileHandleForWriting.write(stdin)
                        try? inPipe.fileHandleForWriting.close()
                    }
                }
                // Drain stderr concurrently so a full pipe can never deadlock the child.
                var errData = Data()
                let group = DispatchGroup()
                group.enter()
                DispatchQueue.global().async {
                    errData = err.fileHandleForReading.readDataToEndOfFile()
                    group.leave()
                }
                let outData = out.fileHandleForReading.readDataToEndOfFile()
                group.wait()
                process.waitUntilExit()
                continuation.resume(returning: ProcessResult(
                    status: process.terminationStatus, stdout: outData, stderr: errData))
            }
        }
    }
}
```

`Sources/MarginSupport/ToolLocator.swift`:

```swift
import Foundation

public struct MissingToolError: Error, CustomStringConvertible, Sendable {
    public let tool: String
    public let installHint: String

    public var description: String { "\(tool) not found. Install it with: \(installHint)" }
}

/// Finds external CLIs. Apps launched from Finder get a minimal PATH, so the
/// usual Homebrew locations are always searched too.
public enum ToolLocator {
    static let fallbackDirectories = ["/opt/homebrew/bin", "/usr/local/bin", "/usr/bin", "/bin"]

    public static func find(
        _ name: String, environment: [String: String] = ProcessInfo.processInfo.environment
    ) -> URL? {
        let pathDirectories = (environment["PATH"] ?? "").split(separator: ":").map(String.init)
        for directory in pathDirectories + fallbackDirectories {
            let candidate = URL(fileURLWithPath: directory).appendingPathComponent(name)
            if FileManager.default.isExecutableFile(atPath: candidate.path) { return candidate }
        }
        return nil
    }

    public static func require(_ name: String, installHint: String) throws -> URL {
        guard let url = find(name) else { throw MissingToolError(tool: name, installHint: installHint) }
        return url
    }
}
```

`Sources/MarginSupport/RealPath.swift`:

```swift
import Foundation

extension String {
    /// Resolves symlinks (e.g. /var -> /private/var) the same way FSEvents and git report paths.
    public func resolvingRealPath() -> String {
        guard let resolved = realpath(self, nil) else { return self }
        defer { free(resolved) }
        return String(cString: resolved)
    }
}
```

```bash
#!/usr/bin/env bash
# Vendors tree-sitter grammars as local C targets and embeds their highlight
# queries as Swift strings (avoids SwiftPM resource bundles in the .app).
set -euo pipefail
cd "$(dirname "$0")/.."
GRAMMARS=("java:Java:v0.23.5" "python:Python:v0.25.0")
QUERIES=Sources/MarginSyntax/HighlightQueries.swift
echo "// Generated by scripts/vendor-grammars.sh — do not edit." > "$QUERIES"
echo "enum HighlightQueries {" >> "$QUERIES"
for entry in "${GRAMMARS[@]}"; do
  IFS=: read -r name Name tag <<< "$entry"
  tmp=$(mktemp -d)
  git -c advice.detachedHead=false clone -q --depth 1 --branch "$tag" "https://github.com/tree-sitter/tree-sitter-$name" "$tmp"
  dest="Sources/TreeSitter$Name"
  rm -rf "$dest"; mkdir -p "$dest/src" "$dest/include"
  cp "$tmp/src/parser.c" "$dest/src/"
  [ -f "$tmp/src/scanner.c" ] && cp "$tmp/src/scanner.c" "$dest/src/"
  cp -R "$tmp/src/tree_sitter" "$dest/src/"
  cp "$tmp/bindings/swift/TreeSitter$Name/"*.h "$dest/include/"
  cp "$tmp/LICENSE" "$dest/LICENSE"
  { echo "    static let $name = #\"\"\""; cat "$tmp/queries/highlights.scm"; echo "\"\"\"#"; } >> "$QUERIES"
  rm -rf "$tmp"
done
echo "}" >> "$QUERIES"
echo "Vendored: ${GRAMMARS[*]}"
```

`Sources/MarginSyntax/SyntaxHighlighter.swift`:

```swift
import Foundation
import SwiftTreeSitter
import TreeSitterJava
import TreeSitterPython

/// A highlighted region. `range` is in UTF-16 units (NSString/NSTextView coordinates);
/// `capture` is the tree-sitter capture name, e.g. "keyword", "string", "function.method".
public struct HighlightSpan: Equatable, Sendable {
    public let range: NSRange
    public let capture: String

    public init(range: NSRange, capture: String) {
        self.range = range
        self.capture = capture
    }
}

public enum SyntaxHighlighter {
    private struct Grammar {
        let language: Language
        let query: Query
    }

    private static let grammars: [SourceLanguage: Grammar] = {
        var result: [SourceLanguage: Grammar] = [:]
        for source in SourceLanguage.allCases {
            let language: Language
            let queryText: String
            switch source {
            case .java:
                language = Language(language: tree_sitter_java())
                queryText = HighlightQueries.java
            case .python:
                language = Language(language: tree_sitter_python())
                queryText = HighlightQueries.python
            }
            if let query = try? Query(language: language, data: Data(queryText.utf8)) {
                result[source] = Grammar(language: language, query: query)
            }
        }
        return result
    }()

    /// Returns spans sorted by location. When several captures hit the same range,
    /// the first one from the query wins (tree-sitter's precedence convention).
    public static func highlights(in source: String, language: SourceLanguage) -> [HighlightSpan] {
        guard let grammar = grammars[language] else { return [] }
        let parser = Parser()
        guard (try? parser.setLanguage(grammar.language)) != nil,
              let tree = parser.parse(source) else { return [] }
        var seen = Set<NSRange>()
        var spans: [HighlightSpan] = []
        for match in grammar.query.execute(in: tree) {
            for capture in match.captures {
                guard let name = capture.name, capture.range.length > 0,
                      seen.insert(capture.range).inserted else { continue }
                spans.append(HighlightSpan(range: capture.range, capture: name))
            }
        }
        return spans.sorted { ($0.range.location, -$0.range.length) < ($1.range.location, -$1.range.length) }
    }
}
```

Note: `tree_sitter_java()` is imported as returning `OpaquePointer!`, and `Language(language:)` accepts it directly.

```swift
import Foundation

public enum ChangeKind: String, Sendable {
    case added, modified, deleted, renamed, untracked, conflicted
}

/// One entry of `git status --porcelain=v2`.
public struct FileStatus: Equatable, Sendable {
    public let path: String
    public let originalPath: String?
    /// Staged state (X) and worktree state (Y), using git's letters ("." = unchanged).
    public let index: Character
    public let worktree: Character
    public let kind: ChangeKind

    public init(path: String, originalPath: String? = nil, index: Character, worktree: Character, kind: ChangeKind) {
        self.path = path
        self.originalPath = originalPath
        self.index = index
        self.worktree = worktree
        self.kind = kind
    }
}

public enum StatusParser {
    /// Parses NUL-separated output of `git status --porcelain=v2 -z --untracked-files=all`.
    public static func parse(_ output: String) -> [FileStatus] {
        var entries = output.split(separator: "\0", omittingEmptySubsequences: true).map(String.init)[...]
        var result: [FileStatus] = []
        while let entry = entries.popFirst() {
            guard let type = entry.first else { continue }
            switch type {
            case "1":
                let fields = entry.split(separator: " ", maxSplits: 8).map(String.init)
                guard fields.count == 9 else { continue }
                let (x, y) = xy(fields[1])
                result.append(FileStatus(path: fields[8], index: x, worktree: y, kind: kind(x, y)))
            case "2":
                let fields = entry.split(separator: " ", maxSplits: 9).map(String.init)
                guard fields.count == 10 else { continue }
                let (x, y) = xy(fields[1])
                let original = entries.popFirst()
                result.append(FileStatus(path: fields[9], originalPath: original, index: x, worktree: y, kind: .renamed))
            case "u":
                let fields = entry.split(separator: " ", maxSplits: 10).map(String.init)
                guard fields.count == 11 else { continue }
                let (x, y) = xy(fields[1])
                result.append(FileStatus(path: fields[10], index: x, worktree: y, kind: .conflicted))
            case "?":
                result.append(FileStatus(path: String(entry.dropFirst(2)), index: "?", worktree: "?", kind: .untracked))
            default:
                continue
            }
        }
        return result
    }

    private static func xy(_ field: String) -> (Character, Character) {
        let chars = Array(field)
        return (chars.first ?? ".", chars.count > 1 ? chars[1] : ".")
    }

    private static func kind(_ x: Character, _ y: Character) -> ChangeKind {
        if x == "A" { return .added }
        if x == "D" || y == "D" { return .deleted }
        return .modified
    }
}
```

```swift
import Foundation

public struct DiffLine: Equatable, Sendable {
    public enum Kind: Sendable { case context, added, removed, noNewline }

    public let kind: Kind
    /// The line as it appears in the patch, including its prefix (" ", "+", "-", "\").
    public let raw: String
    public let oldLine: Int?
    public let newLine: Int?

    /// Line content without the patch prefix.
    public var text: String { String(raw.dropFirst()) }
}

public struct Hunk: Equatable, Sendable, Identifiable {
    public let header: String
    public let oldStart: Int
    public let oldCount: Int
    public let newStart: Int
    public let newCount: Int
    public let lines: [DiffLine]

    public var id: String { header }
    public var additions: Int { lines.filter { $0.kind == .added }.count }
    public var deletions: Int { lines.filter { $0.kind == .removed }.count }
}

public struct FileDiff: Equatable, Sendable, Identifiable {
    /// Lines from `diff --git` up to and including `+++` (or the "Binary files" line).
    public let headerLines: [String]
    public let oldPath: String?
    public let newPath: String?
    public let hunks: [Hunk]
    public let isBinary: Bool

    public var id: String { path }
    public var path: String { newPath ?? oldPath ?? "" }
    public var additions: Int { hunks.reduce(0) { $0 + $1.additions } }
    public var deletions: Int { hunks.reduce(0) { $0 + $1.deletions } }
    public var isNewFile: Bool { oldPath == nil }
    public var isDeletedFile: Bool { newPath == nil }

    /// A patch containing only `hunk` (or all hunks when nil), suitable for `git apply`.
    public func patch(only hunk: Hunk? = nil) -> String {
        let selected = hunk.map { [$0] } ?? hunks
        var lines = headerLines
        for h in selected {
            lines.append(h.header)
            lines.append(contentsOf: h.lines.map(\.raw))
        }
        return lines.joined(separator: "\n") + "\n"
    }
}

public enum DiffParser {
    public static func parse(_ text: String) -> [FileDiff] {
        var files: [FileDiff] = []
        var lines = text.components(separatedBy: "\n")[...]
        if lines.last == "" { lines.removeLast() }

        while let first = lines.first {
            guard first.hasPrefix("diff --git ") else { lines.removeFirst(); continue }
            var header = [lines.removeFirst()]
            var oldPath: String? = gitHeaderPath(first, side: "a/")
            var newPath: String? = gitHeaderPath(first, side: "b/")
            var isBinary = false

            while let line = lines.first, !line.hasPrefix("@@"), !line.hasPrefix("diff --git ") {
                header.append(lines.removeFirst())
                if line.hasPrefix("--- ") { oldPath = patchPath(String(line.dropFirst(4))) }
                else if line.hasPrefix("+++ ") { newPath = patchPath(String(line.dropFirst(4))) }
                else if line.hasPrefix("new file mode") { oldPath = nil }
                else if line.hasPrefix("deleted file mode") { newPath = nil }
                else if line.hasPrefix("Binary files ") { isBinary = true }
            }

            var hunks: [Hunk] = []
            while let line = lines.first, line.hasPrefix("@@") {
                lines.removeFirst()
                guard let (oldStart, oldCount, newStart, newCount) = parseHunkHeader(line) else { continue }
                var body: [DiffLine] = []
                var oldNo = oldStart, newNo = newStart
                while let next = lines.first, !next.hasPrefix("@@"), !next.hasPrefix("diff --git ") {
                    lines.removeFirst()
                    switch next.first {
                    case " ":
                        body.append(DiffLine(kind: .context, raw: next, oldLine: oldNo, newLine: newNo))
                        oldNo += 1; newNo += 1
                    case "-":
                        body.append(DiffLine(kind: .removed, raw: next, oldLine: oldNo, newLine: nil))
                        oldNo += 1
                    case "+":
                        body.append(DiffLine(kind: .added, raw: next, oldLine: nil, newLine: newNo))
                        newNo += 1
                    case "\\":
                        body.append(DiffLine(kind: .noNewline, raw: next, oldLine: nil, newLine: nil))
                    case nil:
                        // An empty context line whose leading space was stripped.
                        body.append(DiffLine(kind: .context, raw: " ", oldLine: oldNo, newLine: newNo))
                        oldNo += 1; newNo += 1
                    default:
                        continue
                    }
                }
                hunks.append(Hunk(header: line, oldStart: oldStart, oldCount: oldCount,
                                  newStart: newStart, newCount: newCount, lines: body))
            }
            files.append(FileDiff(headerLines: header, oldPath: oldPath, newPath: newPath, hunks: hunks, isBinary: isBinary))
        }
        return files
    }

    private static func patchPath(_ value: String) -> String? {
        if value == "/dev/null" { return nil }
        if value.hasPrefix("a/") || value.hasPrefix("b/") { return String(value.dropFirst(2)) }
        return value
    }

    /// Extracts a path from `diff --git a/x b/x`. Used for binary/mode-only diffs without ---/+++.
    private static func gitHeaderPath(_ line: String, side: String) -> String? {
        let rest = line.dropFirst("diff --git ".count)
        guard let split = rest.range(of: " b/", options: .backwards) else { return nil }
        let value = side == "a/" ? rest[..<split.lowerBound] : rest[split.upperBound...]
        return side == "a/" ? String(value.dropFirst(2)) : String(value)
    }

    private static func parseHunkHeader(_ line: String) -> (Int, Int, Int, Int)? {
        // @@ -oldStart[,oldCount] +newStart[,newCount] @@ optional section
        let parts = line.split(separator: " ")
        guard parts.count >= 3 else { return nil }
        func range(_ s: Substring) -> (Int, Int)? {
            let nums = s.dropFirst().split(separator: ",", omittingEmptySubsequences: false)
            guard let start = Int(nums[0]) else { return nil }
            return (start, nums.count > 1 ? Int(nums[1]) ?? 1 : 1)
        }
        guard let old = range(parts[1]), let new = range(parts[2]) else { return nil }
        return (old.0, old.1, new.0, new.1)
    }
}
```

```swift
import Foundation

public struct Worktree: Equatable, Sendable, Identifiable {
    public let path: URL
    public let head: String
    /// Short branch name ("main"), nil when detached or bare.
    public let branch: String?
    public let isBare: Bool

    public var id: String { path.path }
    public var displayName: String { branch ?? "\(path.lastPathComponent) (detached)" }
}

public enum WorktreeParser {
    /// Parses `git worktree list --porcelain`.
    public static func parse(_ output: String) -> [Worktree] {
        output.components(separatedBy: "\n\n").compactMap { block in
            var path: String?, head = "", branch: String?, bare = false
            for line in block.split(separator: "\n") {
                if line.hasPrefix("worktree ") { path = String(line.dropFirst("worktree ".count)) }
                else if line.hasPrefix("HEAD ") { head = String(line.dropFirst("HEAD ".count)) }
                else if line.hasPrefix("branch ") {
                    let ref = line.dropFirst("branch ".count)
                    branch = ref.hasPrefix("refs/heads/") ? String(ref.dropFirst("refs/heads/".count)) : String(ref)
                }
                else if line == "bare" { bare = true }
            }
            guard let path else { return nil }
            return Worktree(path: URL(fileURLWithPath: path), head: head, branch: branch, isBare: bare)
        }
    }
}
```

```swift
import Foundation
import MarginSupport

/// Typed wrapper around the git CLI for one working tree.
public struct Git: Sendable {
    /// Hash of git's empty tree, used as the base when a repo has no commits yet.
    public static let emptyTree = "4b825dc642cb6eb9a060e54bf8d69288fbee4904"

    public let root: URL
    private let executable: URL

    public init(root: URL) throws {
        self.root = root
        self.executable = try ToolLocator.require("git", installHint: "xcode-select --install")
    }

    // MARK: Queries

    public static func isRepository(_ url: URL) async -> Bool {
        guard let git = try? Git(root: url) else { return false }
        return (try? await git.run(["rev-parse", "--is-inside-work-tree"]))?.trimmingCharacters(in: .whitespacesAndNewlines) == "true"
    }

    public func status() async throws -> [FileStatus] {
        StatusParser.parse(try await run(["status", "--porcelain=v2", "-z", "--untracked-files=all"]))
    }

    /// Tracked and untracked (non-ignored) files that exist on disk, repo-relative.
    public func listFiles() async throws -> [String] {
        let output = try await run(["ls-files", "--cached", "--others", "--exclude-standard", "-z"])
        var seen = Set<String>()
        return output.split(separator: "\0").map(String.init).filter { path in
            seen.insert(path).inserted
                && FileManager.default.fileExists(atPath: root.appendingPathComponent(path).path)
        }
    }

    /// HEAD's commit hash, or nil in a repository without commits.
    public func headCommit() async throws -> String? {
        let result = try await ProcessRunner.run(executable, ["-C", root.path, "rev-parse", "--verify", "-q", "HEAD"])
        return result.status == 0 ? result.stdoutString.trimmingCharacters(in: .whitespacesAndNewlines) : nil
    }

    public func mergeBase(_ a: String, _ b: String) async throws -> String {
        try await run(["merge-base", a, b]).trimmingCharacters(in: .whitespacesAndNewlines)
    }

    public func worktrees() async throws -> [Worktree] {
        WorktreeParser.parse(try await run(["worktree", "list", "--porcelain"]))
    }

    /// Diff of `base` (a commit) against the working tree, including untracked files.
    public func diffWorkingTree(against base: String) async throws -> [FileDiff] {
        var diffs = DiffParser.parse(try await run(["diff", "--no-color", "--no-ext-diff", "--no-renames", base]))
        for status in try await status() where status.kind == .untracked {
            diffs.append(contentsOf: DiffParser.parse(try await untrackedDiff(status.path)))
        }
        return diffs.sorted { $0.path < $1.path }
    }

    /// File contents at a revision, or nil if the file does not exist there.
    public func show(path: String, revision: String) async throws -> String? {
        let result = try await ProcessRunner.run(executable, ["-C", root.path, "show", "\(revision):\(path)"])
        return result.status == 0 ? result.stdoutString : nil
    }

    // MARK: Mutations

    /// Applies a patch to the working tree (or to the index when `cached`).
    public func apply(patch: String, reverse: Bool = false, cached: Bool = false) async throws {
        var args = ["apply", "--whitespace=nowarn"]
        if reverse { args.append("-R") }
        if cached { args.append("--cached") }
        try await run(args + ["-"], stdin: Data(patch.utf8))
    }

    public func stage(paths: [String]) async throws {
        try await run(["add", "-A", "--"] + paths)
    }

    public func unstage(paths: [String]) async throws {
        guard try await headCommit() != nil else {
            try await run(["rm", "-q", "--cached", "--ignore-unmatch", "--"] + paths)
            return
        }
        try await run(["reset", "-q", "HEAD", "--"] + paths)
    }

    public func commit(message: String) async throws {
        try await run(["commit", "-q", "-m", message])
    }

    // MARK: Plumbing

    @discardableResult
    func run(_ args: [String], stdin: Data? = nil) async throws -> String {
        let result = try await ProcessRunner.run(
            executable, ["-C", root.path, "-c", "core.quotePath=false"] + args, stdin: stdin)
        guard result.status == 0 else {
            throw ProcessError(command: "git " + args.joined(separator: " "), status: result.status, stderr: result.stderrString)
        }
        return result.stdoutString
    }

    private func untrackedDiff(_ path: String) async throws -> String {
        // --no-index exits with 1 when files differ, which is always the case here.
        let result = try await ProcessRunner.run(
            executable, ["-C", root.path, "diff", "--no-color", "--no-ext-diff", "--no-index", "--", "/dev/null", path])
        guard result.status <= 1 else {
            throw ProcessError(command: "git diff --no-index \(path)", status: result.status, stderr: result.stderrString)
        }
        return result.stdoutString
    }
}
```

```swift
import Foundation

public struct FileNode: Identifiable, Hashable, Sendable {
    /// Repo-relative path; unique within a tree.
    public let id: String
    public let name: String
    /// nil for files, so SwiftUI's OutlineGroup shows no disclosure triangle.
    public let children: [FileNode]?

    public var isDirectory: Bool { children != nil }
}

public enum FileTree {
    /// Builds a sorted tree (directories first, then case-insensitive by name) from relative file paths.
    public static func build(paths: [String]) -> [FileNode] {
        final class Dir {
            var dirs: [String: Dir] = [:]
            var files: Set<String> = []
        }
        let root = Dir()
        for path in paths {
            let parts = path.split(separator: "/").map(String.init)
            guard let fileName = parts.last else { continue }
            var dir = root
            for part in parts.dropLast() {
                if let next = dir.dirs[part] { dir = next } else {
                    let next = Dir()
                    dir.dirs[part] = next
                    dir = next
                }
            }
            dir.files.insert(fileName)
        }
        func nodes(_ dir: Dir, prefix: String) -> [FileNode] {
            let order: (String, String) -> Bool = { $0.localizedStandardCompare($1) == .orderedAscending }
            let dirNodes = dir.dirs.keys.sorted(by: order).map { name in
                FileNode(id: prefix + name, name: name, children: nodes(dir.dirs[name]!, prefix: prefix + name + "/"))
            }
            let fileNodes = dir.files.sorted(by: order).map { FileNode(id: prefix + $0, name: $0, children: nil) }
            return dirNodes + fileNodes
        }
        return nodes(root, prefix: "")
    }
}
```

```swift
import Foundation
import MarginGit

/// An opened folder. Lists files (respecting .gitignore when it is a git repo).
public struct Workspace: Sendable {
    public let root: URL
    public let isGitRepository: Bool

    /// Directories skipped when the folder is not a git repo.
    static let skippedDirectories: Set<String> = [".git", "node_modules", ".build", "build", "target", "__pycache__", ".venv", ".idea"]
    static let maxFiles = 200_000

    public static func open(_ url: URL) async -> Workspace {
        let root = URL(fileURLWithPath: url.path.resolvingRealPath())
        return Workspace(root: root, isGitRepository: await Git.isRepository(root))
    }

    public func listFiles() async throws -> [String] {
        if isGitRepository { return try await Git(root: root).listFiles() }
        return enumerateFiles()
    }

    private func enumerateFiles() -> [String] {
        guard let enumerator = FileManager.default.enumerator(
            at: root, includingPropertiesForKeys: [.isDirectoryKey], options: [.skipsHiddenFiles]) else { return [] }
        let prefixLength = root.path.count + 1
        var result: [String] = []
        for case let url as URL in enumerator {
            let isDirectory = (try? url.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) ?? false
            if isDirectory {
                if Self.skippedDirectories.contains(url.lastPathComponent) { enumerator.skipDescendants() }
                continue
            }
            result.append(String(url.path.resolvingRealPath().dropFirst(prefixLength)))
            if result.count >= Self.maxFiles { break }
        }
        return result.sorted()
    }
}
```

```swift
import CoreServices
import MarginSupport
import Foundation

/// Watches a directory tree with FSEvents and yields batches of changed repo-relative paths.
/// Paths inside `.git/` are included so callers can react to index/HEAD changes.
public final class FileWatcher: @unchecked Sendable {
    public let root: URL
    public let events: AsyncStream<[String]>
    private let continuation: AsyncStream<[String]>.Continuation
    private let rootPath: String
    private var stream: FSEventStreamRef?
    private let queue = DispatchQueue(label: "margin.filewatcher")

    public init(root: URL, latency: TimeInterval = 0.2) {
        self.root = root
        self.rootPath = root.path.resolvingRealPath()
        (events, continuation) = AsyncStream.makeStream(of: [String].self, bufferingPolicy: .unbounded)
        start(latency: latency)
    }

    deinit { stop() }

    public func stop() {
        queue.sync {
            guard let stream else { return }
            FSEventStreamStop(stream)
            FSEventStreamInvalidate(stream)
            FSEventStreamRelease(stream)
            self.stream = nil
        }
        continuation.finish()
    }

    private func start(latency: TimeInterval) {
        var context = FSEventStreamContext(
            version: 0, info: Unmanaged.passUnretained(self).toOpaque(), retain: nil, release: nil, copyDescription: nil)
        let callback: FSEventStreamCallback = { _, info, count, paths, _, _ in
            guard let info else { return }
            let watcher = Unmanaged<FileWatcher>.fromOpaque(info).takeUnretainedValue()
            let array = unsafeBitCast(paths, to: NSArray.self)
            watcher.handle((0..<count).compactMap { array[$0] as? String })
        }
        let flags = UInt32(kFSEventStreamCreateFlagFileEvents | kFSEventStreamCreateFlagUseCFTypes | kFSEventStreamCreateFlagNoDefer)
        guard let stream = FSEventStreamCreate(
            nil, callback, &context, [rootPath] as CFArray,
            FSEventStreamEventId(kFSEventStreamEventIdSinceNow), latency, flags) else {
            continuation.finish()
            return
        }
        self.stream = stream
        FSEventStreamSetDispatchQueue(stream, queue)
        FSEventStreamStart(stream)
    }

    private func handle(_ absolutePaths: [String]) {
        let prefix = rootPath + "/"
        var seen = Set<String>()
        let relative = absolutePaths.compactMap { path -> String? in
            guard path.hasPrefix(prefix) else { return nil }
            let rel = String(path.dropFirst(prefix.count))
            return seen.insert(rel).inserted ? rel : nil
        }
        if !relative.isEmpty { continuation.yield(relative) }
    }
}
```

```swift
import Foundation

/// IntelliJ-style "Go to file" matching: query characters must appear in order;
/// an uppercase query character must hit a word start ("US" → UserService);
/// matches at word starts, in the file name, and consecutive runs score higher.
public enum FuzzyMatcher {
    public static func score(query: String, candidate: String) -> Int? {
        let q = Array(query.filter { !$0.isWhitespace })
        guard !q.isEmpty else { return 0 }
        let original = Array(candidate)
        let lower = Array(candidate.lowercased())
        guard lower.count == original.count else { return nil }
        let fileNameStart = original.lastIndex(of: "/").map { $0 + 1 } ?? 0

        // Prefer matching the query inside the file name: try there first, then the whole path.
        for start in [fileNameStart, 0] {
            if let score = match(q, lower: lower, original: original, from: start, fileNameStart: fileNameStart) {
                return score
            }
        }
        return nil
    }

    private static func match(_ q: [Character], lower: [Character], original: [Character], from start: Int, fileNameStart: Int) -> Int? {
        var score = 0
        var qi = 0
        var previousMatch = -2
        var i = start
        while i < lower.count, qi < q.count {
            let wanted = Character(q[qi].lowercased())
            let wordStart = isWordStart(original, i)
            if lower[i] == wanted, !q[qi].isUppercase || wordStart {
                score += 1
                if i == previousMatch + 1 { score += 5 }
                if wordStart { score += 8 }
                if i >= fileNameStart { score += 2 }
                previousMatch = i
                qi += 1
            }
            i += 1
        }
        guard qi == q.count else { return nil }
        return score * 100 - original.count
    }

    private static func isWordStart(_ chars: [Character], _ i: Int) -> Bool {
        if i == 0 { return true }
        let prev = chars[i - 1]
        if "/_-. ".contains(prev) { return true }
        return chars[i].isUppercase && prev.isLowercase
    }
}

public enum FileFinder {
    /// Returns up to `limit` paths ranked best first.
    public static func search(_ query: String, in paths: [String], limit: Int = 50) -> [String] {
        guard !query.isEmpty else { return Array(paths.prefix(limit)) }
        return paths
            .compactMap { path in FuzzyMatcher.score(query: query, candidate: path).map { (path, $0) } }
            .sorted { $0.1 != $1.1 ? $0.1 > $1.1 : $0.0 < $1.0 }
            .prefix(limit)
            .map(\.0)
    }
}
```

```swift
import Foundation
import MarginSupport

public struct TextMatch: Equatable, Sendable, Identifiable {
    public let path: String
    /// 1-based line number.
    public let line: Int
    public let text: String
    /// Matched ranges as UTF-8 byte offsets into `text`.
    public let byteRanges: [Range<Int>]

    public var id: String { "\(path):\(line)" }
}

/// Full-text search backed by ripgrep (`rg --json`).
public struct TextSearch: Sendable {
    public let root: URL
    public var maxResults = 2_000

    public init(root: URL) { self.root = root }

    public func search(_ query: String, regex: Bool = false) async throws -> [TextMatch] {
        guard !query.isEmpty else { return [] }
        let rg = try ToolLocator.require("rg", installHint: "brew install ripgrep")
        var args = ["--json", "--smart-case", "--max-columns", "500", "--max-columns-preview"]
        if !regex { args.append("--fixed-strings") }
        args += ["--", query, "."]
        let result = try await ProcessRunner.run(rg, args, cwd: root)
        // ripgrep exits 1 when nothing matched.
        guard result.status <= 1 else {
            throw ProcessError(command: "rg", status: result.status, stderr: result.stderrString)
        }
        return Self.parse(result.stdoutString, limit: maxResults)
    }

    static func parse(_ output: String, limit: Int) -> [TextMatch] {
        struct Event: Decodable {
            struct Text: Decodable { let text: String? }
            struct Submatch: Decodable { let start: Int; let end: Int }
            struct DataPayload: Decodable {
                let path: Text?
                let lines: Text?
                let line_number: Int?
                let submatches: [Submatch]?
            }
            let type: String
            let data: DataPayload
        }
        let decoder = JSONDecoder()
        var matches: [TextMatch] = []
        for line in output.split(separator: "\n") {
            guard matches.count < limit,
                  let event = try? decoder.decode(Event.self, from: Data(line.utf8)),
                  event.type == "match",
                  var path = event.data.path?.text,
                  let number = event.data.line_number,
                  let text = event.data.lines?.text else { continue }
            if path.hasPrefix("./") { path.removeFirst(2) }
            let trimmed = text.hasSuffix("\n") ? String(text.dropLast()) : text
            let ranges = (event.data.submatches ?? []).map { $0.start..<$0.end }
            matches.append(TextMatch(path: path, line: number, text: trimmed, byteRanges: ranges))
        }
        return matches
    }
```

```swift
import Foundation
import MarginGit

/// What is being reviewed.
public enum ReviewTarget: Hashable, Sendable, Identifiable {
    /// Uncommitted changes (staged + unstaged + untracked) in a checkout, vs HEAD.
    case workingTree(root: URL)
    /// Everything an agent did in a worktree (commits + uncommitted), vs the merge-base with `baseBranch`.
    case worktree(root: URL, branch: String, baseBranch: String)

    public var id: String { root.path }

    public var root: URL {
        switch self {
        case .workingTree(let root), .worktree(let root, _, _): return root
        }
    }

    /// Staging/committing only makes sense against HEAD of the reviewed checkout.
    public var allowsStaging: Bool {
        if case .workingTree = self { return true }
        return false
    }

    public var title: String {
        switch self {
        case .workingTree(let root): return "\(root.lastPathComponent) — uncommitted"
        case .worktree(_, let branch, let base): return "\(branch) vs \(base)"
        }
    }

    /// One target per worktree: the main checkout reviews uncommitted work,
    /// every other worktree reviews its branch against the main checkout's branch.
    public static func all(from worktrees: [Worktree]) -> [ReviewTarget] {
        let usable = worktrees.filter { !$0.isBare }
        guard let main = usable.first else { return [] }
        let base = main.branch ?? main.head
        return [.workingTree(root: main.path)] + usable.dropFirst().map {
            .worktree(root: $0.path, branch: $0.displayName, baseBranch: base)
        }
    }
}
```

```swift
import Foundation
import MarginGit

public struct ChangedFile: Identifiable, Equatable, Sendable {
    public let diff: FileDiff
    public let kind: ChangeKind

    public var id: String { diff.path }
    public var path: String { diff.path }
}

public struct ReviewError: Error, CustomStringConvertible, Sendable {
    public let description: String
}

/// Computes and acts on the changes of a review target. Stateless: the disk and git are the source of truth.
public struct ReviewEngine: Sendable {
    public let target: ReviewTarget
    private let git: Git

    public init(target: ReviewTarget) throws {
        self.target = target
        self.git = try Git(root: target.root)
    }

    /// The commit the target is compared against.
    public func baseRevision() async throws -> String {
        switch target {
        case .workingTree:
            return try await git.headCommit() ?? Git.emptyTree
        case .worktree(_, _, let baseBranch):
            return try await git.mergeBase(baseBranch, "HEAD")
        }
    }

    public func changes() async throws -> [ChangedFile] {
        let diffs = try await git.diffWorkingTree(against: try await baseRevision())
        return diffs.map { diff in
            let kind: ChangeKind = diff.isNewFile ? .added : diff.isDeletedFile ? .deleted : .modified
            return ChangedFile(diff: diff, kind: kind)
        }
    }

    /// Old (base) and new (working tree) contents, for side-by-side display.
    public func contents(of file: ChangedFile) async throws -> (old: String?, new: String?) {
        var old: String?
        if let oldPath = file.diff.oldPath {
            old = try await git.show(path: oldPath, revision: try await baseRevision())
        }
        let new = file.diff.newPath.flatMap { try? String(contentsOf: target.root.appendingPathComponent($0), encoding: .utf8) }
        return (old, new)
    }

    // MARK: Actions

    public func revert(_ hunk: Hunk, in file: ChangedFile) async throws {
        try requireText(file)
        try await git.apply(patch: file.diff.patch(only: hunk), reverse: true)
    }

    public func revert(_ file: ChangedFile) async throws {
        try requireText(file)
        try await git.apply(patch: file.diff.patch(), reverse: true)
        // Drop any staged version too, so the file is fully back to base.
        if target.allowsStaging { try await git.unstage(paths: [file.path]) }
    }

    public func stage(_ hunk: Hunk, in file: ChangedFile) async throws {
        try requireStaging()
        try requireText(file)
        try await git.apply(patch: file.diff.patch(only: hunk), cached: true)
    }

    public func stage(_ file: ChangedFile) async throws {
        try requireStaging()
        try await git.stage(paths: [file.path])
    }

    public func commit(message: String) async throws {
        try requireStaging()
        guard !message.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else {
            throw ReviewError(description: "Commit message is empty.")
        }
        try await git.commit(message: message)
    }

    private func requireStaging() throws {
        guard target.allowsStaging else {
            throw ReviewError(description: "Staging is only available when reviewing uncommitted changes.")
        }
    }

    private func requireText(_ file: ChangedFile) throws {
        guard !file.diff.isBinary else { throw ReviewError(description: "\(file.path) is binary; revert it with git.") }
    }
}
```

```swift
import MarginGit

public struct SideBySideRow: Equatable, Sendable {
    public let left: DiffLine?
    public let right: DiffLine?
}

public enum SideBySide {
    /// Aligns a hunk into two columns: context on both sides, removed lines paired with added lines.
    public static func rows(for hunk: Hunk) -> [SideBySideRow] {
        var rows: [SideBySideRow] = []
        var removed: [DiffLine] = []
        var added: [DiffLine] = []
        func flush() {
            for i in 0..<max(removed.count, added.count) {
                rows.append(SideBySideRow(left: i < removed.count ? removed[i] : nil, right: i < added.count ? added[i] : nil))
            }
            removed.removeAll()
            added.removeAll()
        }
        for line in hunk.lines {
            switch line.kind {
            case .removed: removed.append(line)
            case .added: added.append(line)
            case .context:
                flush()
                rows.append(SideBySideRow(left: line, right: line))
            case .noNewline:
                continue
            }
        }
        flush()
        return rows
    }
}
```

```swift
import Foundation

/// Files touched during this session, most recent first — lets you follow an agent as it works.
public struct ChangeTimeline: Sendable {
    public struct Entry: Equatable, Sendable, Identifiable {
        public let path: String
        public let date: Date
        public var id: String { path }
    }

    public private(set) var entries: [Entry] = []
    public var liveWindow: TimeInterval = 3
    public var maxEntries = 200

    public init() {}

    /// Records changed paths. Paths inside `.git/` are ignored.
    public mutating func record(_ paths: [String], at date: Date = Date()) {
        let relevant = paths.filter { !$0.hasPrefix(".git/") && $0 != ".git" }
        guard !relevant.isEmpty else { return }
        let touched = Set(relevant)
        entries.removeAll { touched.contains($0.path) }
        entries.insert(contentsOf: relevant.sorted().map { Entry(path: $0, date: date) }, at: 0)
        if entries.count > maxEntries { entries.removeLast(entries.count - maxEntries) }
    }

    /// True when something changed within the live window — an agent is probably working.
    public func isLive(now: Date = Date()) -> Bool {
        guard let latest = entries.first else { return false }
        return now.timeIntervalSince(latest.date) <= liveWindow
    }
}
```

```swift
    func isFile(_ path: String) -> Bool { fileSet.contains(path) }

    // MARK: Workspace

    /// `Margin -workspace /path/to/repo` (AppKit maps `-key value` arguments into UserDefaults;
    /// a bare path argument would be treated as an open-document event and suppress the window).
    func openFromCommandLine() async {
        guard workspace == nil, let path = UserDefaults.standard.string(forKey: "workspace") else { return }
        var isDirectory: ObjCBool = false
        if FileManager.default.fileExists(atPath: path, isDirectory: &isDirectory), isDirectory.boolValue {
            await open(URL(fileURLWithPath: path))
        }
    }

    func chooseFolder() {
        let panel = NSOpenPanel()
        panel.canChooseDirectories = true
        panel.canChooseFiles = false
        panel.prompt = "Open"
        guard panel.runModal() == .OK, let url = panel.url else { return }
        Task { await open(url) }
    }

    func open(_ url: URL) async {
        saveAll()
        let opened = await Workspace.open(url)
        workspace = opened
        documents = []
        activeDocumentID = nil
        await reloadFiles()
        startWatching(opened.root)
    }

    private func reloadFiles() async {
        guard let workspace else { return }
        do {
            files = try await workspace.listFiles()
            fileSet = Set(files)
            tree = FileTree.build(paths: files)
        } catch {
            report(error)
        }
    }

    private func startWatching(_ root: URL) {
        watchTask?.cancel()
        watcher?.stop()
        let watcher = FileWatcher(root: root)
        self.watcher = watcher
        watchTask = Task { [weak self] in
            for await batch in watcher.events {
                await self?.handleChanges(batch)
            }
        }
    }

    private func handleChanges(_ paths: [String]) async {
        let workingFiles = paths.filter { !$0.hasPrefix(".git/") && $0 != ".git" }
        let rootPath = workspace?.root.path ?? ""
        let treeChanged = workingFiles.contains { path in
            fileSet.contains(path) != FileManager.default.fileExists(atPath: rootPath + "/" + path)
        }
        if treeChanged { await reloadFiles() }
        for document in documents where workingFiles.contains(document.id) {
            document.diskChanged()
        }
    }

    // MARK: Documents

    func openFile(_ path: String, line: Int? = nil) {
        if let existing = documents.first(where: { $0.id == path }) {
            existing.pendingLine = line
            activeDocumentID = path
            return
        }
        guard let workspace else { return }
        do {
            let document = try EditorDocument(root: workspace.root, path: path)
            document.pendingLine = line
            documents.append(document)
            activeDocumentID = path
        } catch {
            report(error)
        }
    }

    func close(_ document: EditorDocument) {
        document.saveIfSafe(report: report)
        documents.removeAll { $0.id == document.id }
        if activeDocumentID == document.id { activeDocumentID = documents.last?.id }
    }

    /// IntelliJ-style autosave: dirty documents are saved unless they conflict with a disk change.
    func saveAll() {
        for document in documents { document.saveIfSafe(report: report) }
    }

    func report(_ error: Error) {
        errorMessage = String(describing: error)
    }
}
```

```swift
@MainActor
final class EditorDocument: ObservableObject, Identifiable {
    /// Repo-relative path.
    let id: String
    let url: URL
    let language: SourceLanguage?
    @Published var text: String
    @Published private(set) var savedText: String
    /// The file changed on disk while this document had unsaved edits.
    @Published private(set) var changedOnDisk = false
    /// A line to reveal next time the editor updates (1-based).
    @Published var pendingLine: Int?

    init(root: URL, path: String) throws {
        id = path
        url = root.appendingPathComponent(path)
        language = SourceLanguage.detect(path: path)
        let content = try String(contentsOf: url, encoding: .utf8)
        text = content
        savedText = content
    }

    var name: String { url.lastPathComponent }
    var isDirty: Bool { text != savedText }

    func save() throws {
        try text.write(to: url, atomically: true, encoding: .utf8)
        savedText = text
        changedOnDisk = false
    }

    func saveIfSafe(report: (Error) -> Void) {
        guard isDirty, !changedOnDisk else { return }
        do { try save() } catch { report(error) }
    }

    /// Called when the file changed on disk (e.g. an agent edited it).
    func diskChanged() {
        guard let disk = try? String(contentsOf: url, encoding: .utf8), disk != savedText else { return }
        if isDirty {
            changedOnDisk = true
        } else {
            text = disk
            savedText = disk
        }
    }

    func reloadFromDisk() {
        if let disk = try? String(contentsOf: url, encoding: .utf8) {
            text = disk
            savedText = disk
        }
        changedOnDisk = false
    }

    func keepMine() {
        changedOnDisk = false
    }
}
```

```swift
enum Theme {
    static let font = NSFont.monospacedSystemFont(ofSize: 13, weight: .regular)

    static func color(for capture: String) -> NSColor {
        switch capture.split(separator: ".").first.map(String.init) ?? capture {
        case "keyword", "include", "conditional", "repeat", "exception": return .systemOrange
        case "string", "character", "escape": return .systemGreen
        case "number", "float", "boolean": return .systemBlue
        case "comment": return .secondaryLabelColor
        case "type", "constructor": return .systemTeal
        case "function", "method": return .systemIndigo
        case "constant": return .systemPurple
        case "attribute", "annotation": return .systemBrown
        default: return .labelColor
        }
    }

    /// Highlights a whole file and returns one AttributedString per line (index = line - 1).
    static func highlightedLines(_ text: String?, path: String) -> [AttributedString] {
        guard let text else { return [] }
        let ns = text as NSString
        var colors = [NSColor?](repeating: nil, count: ns.length)
        if let language = SourceLanguage.detect(path: path) {
            for span in SyntaxHighlighter.highlights(in: text, language: language) {
                let color = color(for: span.capture)
                for i in span.range.location..<min(NSMaxRange(span.range), ns.length) { colors[i] = color }
            }
        }
        var lines: [AttributedString] = []
        var lineStart = 0
        while lineStart <= ns.length {
            let lineRange = ns.lineRange(for: NSRange(location: lineStart, length: 0))
            var contentEnd = NSMaxRange(lineRange)
            while contentEnd > lineRange.location, [10, 13].contains(ns.character(at: contentEnd - 1)) { contentEnd -= 1 }
            var line = AttributedString()
            var runStart = lineRange.location
            while runStart < contentEnd {
                var runEnd = runStart + 1
                while runEnd < contentEnd, colors[runEnd] == colors[runStart] { runEnd += 1 }
                var piece = AttributedString(ns.substring(with: NSRange(location: runStart, length: runEnd - runStart)))
                if let color = colors[runStart] { piece.foregroundColor = Color(nsColor: color) }
                line += piece
                runStart = runEnd
            }
            lines.append(line)
            if NSMaxRange(lineRange) == lineStart { break }
            lineStart = NSMaxRange(lineRange)
            if lineStart == ns.length { break }
        }
        return lines
    }
}
```

```swift
/// Plain-text code editor (NSTextView) with tree-sitter highlighting.
struct CodeTextView: NSViewRepresentable {
    @ObservedObject var document: EditorDocument

    func makeCoordinator() -> Coordinator { Coordinator(document: document) }

    func makeNSView(context: Context) -> NSScrollView {
        let scrollView = NSTextView.scrollableTextView()
        let textView = scrollView.documentView as! NSTextView
        textView.isRichText = false
        textView.allowsUndo = true
        textView.font = Theme.font
        textView.isAutomaticQuoteSubstitutionEnabled = false
        textView.isAutomaticDashSubstitutionEnabled = false
        textView.isAutomaticSpellingCorrectionEnabled = false
        textView.isContinuousSpellCheckingEnabled = false
        textView.textContainerInset = NSSize(width: 6, height: 6)
        textView.string = document.text
        textView.delegate = context.coordinator
        context.coordinator.textView = textView
        context.coordinator.highlight()
        return scrollView
    }

    func updateNSView(_ scrollView: NSScrollView, context: Context) {
        let coordinator = context.coordinator
        guard let textView = coordinator.textView else { return }
        if textView.string != document.text {
            // The document was reloaded from disk.
            let selection = textView.selectedRange()
            textView.string = document.text
            textView.setSelectedRange(NSRange(location: min(selection.location, (document.text as NSString).length), length: 0))
            coordinator.highlight()
        }
        if let line = document.pendingLine {
            coordinator.reveal(line: line)
            DispatchQueue.main.async { document.pendingLine = nil }
        }
    }

    @MainActor
    final class Coordinator: NSObject, NSTextViewDelegate {
        let document: EditorDocument
        weak var textView: NSTextView?
        private var pendingHighlight: DispatchWorkItem?

        init(document: EditorDocument) { self.document = document }

        func textDidChange(_ notification: Notification) {
            guard let textView else { return }
            document.text = textView.string
            pendingHighlight?.cancel()
            let work = DispatchWorkItem { [weak self] in self?.highlight() }
            pendingHighlight = work
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.15, execute: work)
        }

        func highlight() {
            guard let textView, let storage = textView.textStorage else { return }
            let full = NSRange(location: 0, length: storage.length)
            storage.beginEditing()
            storage.setAttributes([.font: Theme.font, .foregroundColor: NSColor.labelColor], range: full)
            if let language = document.language {
                for span in SyntaxHighlighter.highlights(in: textView.string, language: language)
                where NSMaxRange(span.range) <= full.length {
                    storage.addAttribute(.foregroundColor, value: Theme.color(for: span.capture), range: span.range)
                }
            }
            storage.endEditing()
            textView.typingAttributes = [.font: Theme.font, .foregroundColor: NSColor.labelColor]
        }

        func reveal(line: Int) {
            guard let textView else { return }
            let text = textView.string as NSString
            var location = 0
            for _ in 1..<max(line, 1) {
                guard location < text.length else { break }
                location = NSMaxRange(text.lineRange(for: NSRange(location: location, length: 0)))
            }
            let range = text.lineRange(for: NSRange(location: min(location, text.length), length: 0))
            textView.scrollRangeToVisible(range)
            textView.setSelectedRange(NSRange(location: range.location, length: 0))
            textView.showFindIndicator(for: range)
            textView.window?.makeFirstResponder(textView)
        }
    }
}
```

```swift
        let query = query, regex = regex
        searchTask = Task {
            do {
                let found = try await TextSearch(root: root).search(query, regex: regex)
                guard !Task.isCancelled else { return }
                results = found
                message = found.isEmpty ? "No matches." : "\(found.count) matches"
            } catch {
                results = []
                message = String(describing: error)
            }
        }
    }

    /// Bolds the matched byte ranges.
    static func highlighted(_ match: TextMatch) -> AttributedString {
        let bytes = Array(match.text.utf8)
        var result = AttributedString()
        var cursor = 0
        for range in match.byteRanges where range.lowerBound >= cursor && range.upperBound <= bytes.count {
            result += AttributedString(String(decoding: bytes[cursor..<range.lowerBound], as: UTF8.self))
            var hit = AttributedString(String(decoding: bytes[range], as: UTF8.self))
            hit.font = .system(.body, design: .monospaced).bold()
            hit.backgroundColor = .yellow.opacity(0.35)
            result += hit
            cursor = range.upperBound
        }
        result += AttributedString(String(decoding: bytes[cursor...], as: UTF8.self))
        return result
    }
```

```swift
        selectedFileID = nil
        timeline = ChangeTimeline()
        guard workspace.isGitRepository else {
            targets = []
            target = nil
            return
        }
        await reloadTargets()
    }

    /// Called with every change batch from the workspace watcher.
    func workspaceChanged(_ paths: [String]) async {
        if paths.contains(where: { $0.hasPrefix(".git/worktrees") }) { await reloadTargets() }
    }

    private func reloadTargets() async {
        guard let root = workspaceRoot else { return }
        do {
            let all = ReviewTarget.all(from: try await Git(root: root).worktrees())
            targets = all
            if let current = target, all.contains(current) { return }
            target = all.first { $0.root.path == root.path } ?? all.first
        } catch {
            onError(error)
        }
    }

    private func targetChanged() {
        selectedFileID = nil
        timeline = ChangeTimeline()
        targetWatchTask?.cancel()
        targetWatcher?.stop()
        targetWatcher = nil
        guard let target else {
            changes = []
            return
        }
        // Paths of other worktrees nested inside this one (e.g. .claude/worktrees/x) are not ours.
        let nested = targets.compactMap { other -> String? in
            let prefix = target.root.path + "/"
            return other != target && other.root.path.hasPrefix(prefix) ? String(other.root.path.dropFirst(prefix.count)) + "/" : nil
        }
        let watcher = FileWatcher(root: target.root)
        targetWatcher = watcher
        targetWatchTask = Task { [weak self] in
            for await batch in watcher.events {
                let own = batch.filter { path in !nested.contains { path.hasPrefix($0) } }
                guard !own.isEmpty else { continue }
                self?.timeline.record(own)
                self?.isLive = self?.timeline.isLive() ?? false
                self?.scheduleRefresh()
            }
        }
        scheduleRefresh()
    }

    func scheduleRefresh() {
        refreshTask?.cancel()
        refreshTask = Task {
            try? await Task.sleep(for: .milliseconds(150))
            guard !Task.isCancelled else { return }
            await refresh()
        }
    }

    func refresh() async {
        guard let engine else {
            changes = []
            return
        }
        do {
            let latest = try await engine.changes()
            if latest != changes { changes = latest }
        } catch {
            onError(error)
        }
    }

    func contents(of file: ChangedFile) async -> (old: String?, new: String?) {
        guard let engine else { return (nil, nil) }
        do { return try await engine.contents(of: file) } catch {
            onError(error)
            return (nil, nil)
        }
    }

    // MARK: Actions

    func revert(_ hunk: Hunk, in file: ChangedFile) { perform { try await $0.revert(hunk, in: file) } }
    func revert(_ file: ChangedFile) { perform { try await $0.revert(file) } }
    func stage(_ hunk: Hunk, in file: ChangedFile) { perform { try await $0.stage(hunk, in: file) } }
    func stage(_ file: ChangedFile) { perform { try await $0.stage(file) } }

    func commit() {
        let message = commitMessage
        perform { [weak self] engine in
            try await engine.commit(message: message)
            self?.commitMessage = ""
        }
    }

    private func perform(_ action: @escaping (ReviewEngine) async throws -> Void) {
        guard let engine else { return }
        Task {
            do { try await action(engine) } catch { onError(error) }
            await refresh()
        }
    }
}
```

```swift
    @Published private(set) var documents: [EditorDocument] = []

    @Published var sidebarMode: SidebarMode = .project

    @Published var showFindInFiles = false
    @Published var errorMessage: String?

    let review = ReviewModel()
    private var fileSet: Set<String> = []
    private var watcher: FileWatcher?
    private var watchTask: Task<Void, Never>?

    init() {
        review.onError = { [weak self] in self?.report($0) }
        NotificationCenter.default.addObserver(
            forName: NSApplication.didResignActiveNotification, object: nil, queue: .main
        ) { [weak self] _ in
            MainActor.assumeIsolated { self?.saveAll() }
        }
    }

    var activeDocument: EditorDocument? { documents.first { $0.id == activeDocumentID } }

    func isFile(_ path: String) -> Bool { fileSet.contains(path) }

    // MARK: Workspace

    /// `Margin -workspace /path/to/repo` (AppKit maps `-key value` arguments into UserDefaults;
    /// a bare path argument would be treated as an open-document event and suppress the window).
    func openFromCommandLine() async {
        guard workspace == nil, let path = UserDefaults.standard.string(forKey: "workspace") else { return }
        var isDirectory: ObjCBool = false
        if FileManager.default.fileExists(atPath: path, isDirectory: &isDirectory), isDirectory.boolValue {
            await open(URL(fileURLWithPath: path))
        }
    }

    func chooseFolder() {
        let panel = NSOpenPanel()
        panel.canChooseDirectories = true
        panel.canChooseFiles = false
        panel.prompt = "Open"
        guard panel.runModal() == .OK, let url = panel.url else { return }
        Task { await open(url) }
    }

    func open(_ url: URL) async {
        saveAll()
        let opened = await Workspace.open(url)
        workspace = opened
        documents = []
        activeDocumentID = nil
        await reloadFiles()
        startWatching(opened.root)
        await review.load(workspace: opened)
    }

    private func reloadFiles() async {
        guard let workspace else { return }
        do {
            files = try await workspace.listFiles()
            fileSet = Set(files)
            tree = FileTree.build(paths: files)
        } catch {
            report(error)
        }
    }

    private func startWatching(_ root: URL) {
        watchTask?.cancel()
        watcher?.stop()
        let watcher = FileWatcher(root: root)
        self.watcher = watcher
        watchTask = Task { [weak self] in
            for await batch in watcher.events {
                await self?.handleChanges(batch)
            }
        }
    }

    private func handleChanges(_ paths: [String]) async {
        let workingFiles = paths.filter { !$0.hasPrefix(".git/") && $0 != ".git" }
        let rootPath = workspace?.root.path ?? ""
        let treeChanged = workingFiles.contains { path in
            fileSet.contains(path) != FileManager.default.fileExists(atPath: rootPath + "/" + path)
        }
        if treeChanged { await reloadFiles() }
        for document in documents where workingFiles.contains(document.id) {
            document.diskChanged()
        }
        await review.workspaceChanged(paths)
    }

    // MARK: Documents

    func openFile(_ path: String, line: Int? = nil) {
        if let existing = documents.first(where: { $0.id == path }) {
            existing.pendingLine = line
            activeDocumentID = path
            return
        }
        guard let workspace else { return }
        do {
            let document = try EditorDocument(root: workspace.root, path: path)
            document.pendingLine = line
            documents.append(document)
            activeDocumentID = path
        } catch {
            report(error)
        }
    }

    func close(_ document: EditorDocument) {
        document.saveIfSafe(report: report)
        documents.removeAll { $0.id == document.id }
        if activeDocumentID == document.id { activeDocumentID = documents.last?.id }
    }

    /// IntelliJ-style autosave: dirty documents are saved unless they conflict with a disk change.
    func saveAll() {
        for document in documents { document.saveIfSafe(report: report) }
    }

    func report(_ error: Error) {
        errorMessage = String(describing: error)
    }
}
```

```swift
final class AppDelegate: NSObject, NSApplicationDelegate {
    private var lastShiftRelease: TimeInterval = 0
    private var shiftWasDown = false

    func applicationDidFinishLaunching(_ notification: Notification) {
        // Needed when running the bare executable (swift run) instead of the .app bundle.
        NSApp.setActivationPolicy(.regular)
        NSApp.activate(ignoringOtherApps: true)
        // IntelliJ's "double Shift" opens Go to File.
        NSEvent.addLocalMonitorForEvents(matching: .flagsChanged) { [weak self] event in
            self?.trackShift(event)
            return event

    private func trackShift(_ event: NSEvent) {
        let flags = event.modifierFlags.intersection(.deviceIndependentFlagsMask)
        let shiftOnly = flags == .shift
        if shiftOnly {
            shiftWasDown = true
        } else if shiftWasDown && flags.isEmpty {
            shiftWasDown = false
            if event.timestamp - lastShiftRelease < 0.35 {
                lastShiftRelease = 0
                NotificationCenter.default.post(name: .marginDoubleShift, object: nil)
            } else {
                lastShiftRelease = event.timestamp
            }
        } else {
            shiftWasDown = false
        }
    }
```

```swift
struct FindInFilesView: View {

            }
            if let message {
                Text(message).foregroundStyle(.secondary)
            }
            List(results) { match in
                VStack(alignment: .leading, spacing: 2) {
                    Text("\(match.path):\(match.line)").font(.caption).foregroundStyle(.secondary)
                    Text(Self.highlighted(match)).font(.system(.body, design: .monospaced)).lineLimit(1)
                }
                .frame(maxWidth: .infinity, alignment: .leading)
                .contentShape(Rectangle())
                .onTapGesture {

                    app.sidebarMode = .project
                    dismiss()
                }
            }
        }
        .padding()
        .frame(width: 760, height: 480)
    }

    private func search() {
        guard let root = app.workspace?.root else { return }
        searchTask?.cancel()
        message = "Searching…"
        let query = query, regex = regex
        searchTask = Task {
            do {
                let found = try await TextSearch(root: root).search(query, regex: regex)
                guard !Task.isCancelled else { return }
                results = found
                message = found.isEmpty ? "No matches." : "\(found.count) matches"
            } catch {
                results = []
                message = String(describing: error)
            }
        }
    }

    /// Bolds the matched byte ranges.
    static func highlighted(_ match: TextMatch) -> AttributedString {
        let bytes = Array(match.text.utf8)
        var result = AttributedString()
        var cursor = 0
        for range in match.byteRanges where range.lowerBound >= cursor && range.upperBound <= bytes.count {
            result += AttributedString(String(decoding: bytes[cursor..<range.lowerBound], as: UTF8.self))
            var hit = AttributedString(String(decoding: bytes[range], as: UTF8.self))
            hit.font = .system(.body, design: .monospaced).bold()
            hit.backgroundColor = .yellow.opacity(0.35)
            result += hit
            cursor = range.upperBound
        }
        result += AttributedString(String(decoding: bytes[cursor...], as: UTF8.self))
        return result
    }
```
